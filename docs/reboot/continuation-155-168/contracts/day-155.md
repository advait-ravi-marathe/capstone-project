```python
import unittest

def only_synthetic(rows):
    for row in rows:
        try:
            if row["source"] != "synthetic":
                raise ValueError("source must be synthetic")
        except (KeyError, ValueError):
            return False
    return True

checks = [{"source": "synthetic"}, {"source": "synthetic"}]
unittest.TestCase().assertTrue(only_synthetic(checks))
print("readiness bridge passed")
```
